package main

import (
 "crypto/sha256"
 "encoding/json"
 "fmt"
 "os"
 "os/exec"
 "strings"
)
func main(){
 dir:="tools/benchmark/research/v0.22-child-storage/";names:=map[string]bool{}
 for _,args:=range [][]string{{"diff","--name-only"},{"ls-files","--others","--exclude-standard"}} {b,e:=exec.Command("git",args...).Output();if e!=nil{panic(e)};for _,p:=range strings.Fields(string(b)){if strings.HasSuffix(p,".go"){names[p]=true}}}
 if len(names)!=5 {panic(fmt.Sprint("unexpected overlay ",names))}
 files,hashes:=map[string]string{},map[string]string{}
 for p:=range names {b,e:=os.ReadFile(p);if e!=nil{panic(e)};files[p]=string(b);hashes[p]=fmt.Sprintf("%x",sha256.Sum256(b))}
 for p,v:=range map[string]any{dir+"candidate-source.json":map[string]any{"base_tag":"v0.21","files":files},dir+"source-hashes.json":hashes}{if _,e:=os.Stat(p);!os.IsNotExist(e){panic("refuse overwrite "+p)};b,e:=json.MarshalIndent(v,"","  ");if e!=nil{panic(e)};if e=os.WriteFile(p,append(b,'\n'),0644);e!=nil{panic(e)}}
}

package main

import (
	"encoding/json"
	"os"
	"testing"

	"github.com/corentings/chess/v2"
)

func TestTrainingQuietMateProof(t *testing.T) {
	data, err := os.ReadFile(os.Getenv("QUIET_PROOF_INPUT"))
	if err != nil { t.Fatal(err) }
	var input struct { Rows []struct {
		ID, Choice string
		Witness struct { FEN string }
		Mates []struct { Move string } `json:"side_to_move_mates_in_one"`
	} }
	if err := json.Unmarshal(data, &input); err != nil { t.Fatal(err) }
	count := 0
	for _, row := range input.Rows {
		if len(row.Mates) == 0 { continue }
		option, err := chess.FEN(row.Witness.FEN)
		if err != nil { t.Fatal(err) }
		p := chess.NewGame(option).Position()
		if inCheck(p) { t.Fatal("expected non-checked static leaf", row.ID) }
		want := map[string]bool{}
		for _, m := range row.Mates { want[m.Move] = true }
		for _, m := range p.ValidMoves() {
			if p.Update(&m).SearchStatus() != chess.Checkmate { continue }
			if !want[m.String()] { t.Fatal("additional mate", row.ID, m) }
			if m.HasTag(chess.Capture) || m.HasTag(chess.EnPassant) || m.Promo() != chess.NoPieceType {
				t.Fatal("expected quiet mating move", row.ID, m)
			}
			delete(want, m.String())
		}
		if len(want) != 0 { t.Fatal("missing mates", row.ID, want) }
		quiet := quiescence(p, quiescenceDepth, -searchInfinity, searchInfinity)
		static := staticScore(p, p.Turn())
		if quiet != static { t.Fatal("stand-pat reconstruction mismatch", row.ID, quiet, static) }
		t.Logf("%s %s: legal quiet mate verified; released full-window quiescence/static=%d", row.ID, row.Choice, quiet)
		count++
	}
	if count != 15 { t.Fatal("unexpected proof count", count) }
}

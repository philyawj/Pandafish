package main

import "math/bits"

const evalFileA uint64 = 0x8080808080808080
const evalFileH uint64 = 0x0101010101010101

// Include the first blocker, matching geometric attacks/defences. This is not
// legal mobility: pinned attackers and prospective check destinations are hints.
func evaluationSliders(square int, occupied uint64, diagonal, straight bool) uint64 {
	var result uint64
	for direction, ray := range exchangeGeometry.rays[square] {
		if (direction < 4 && !straight) || (direction >= 4 && !diagonal) {
			continue
		}
		hits := ray & occupied
		if hits != 0 {
			first := 63 - bits.TrailingZeros64(hits)
			if direction%2 == 0 {
				first = bits.LeadingZeros64(hits)
			}
			ray &^= exchangeGeometry.rays[first][direction]
		}
		result |= ray
	}
	return result
}

func evaluationPawnAttacks(pawns uint64, side int) uint64 {
	if side == 0 {
		return (pawns &^ evalFileH >> 9) | (pawns &^ evalFileA >> 7)
	}
	return (pawns &^ evalFileH << 7) | (pawns &^ evalFileA << 9)
}

// Return a White-minus-Black positional correction. No legal list or heap
// storage is built, and no feature can approach terminal mate scores.
func checkingPressure(pieces [12]uint64, phase int) int {
	var occupancy [2]uint64
	for i, set := range pieces {
		occupancy[i/6] |= set
	}
	occupied := occupancy[0] | occupancy[1]
	var attacks [2][6]uint64
	var all, twice [2]uint64
	var kings [2]int
	var rings [2]uint64
	for side := 0; side < 2; side++ {
		if bits.OnesCount64(pieces[side*6]) != 1 {
			return 0
		}
		kings[side] = bits.LeadingZeros64(pieces[side*6])
		rings[side] = exchangeGeometry.kings[kings[side]]
	}
	var ringUnits, attackers [2]int
	for i, set := range pieces {
		side, kind := i/6, i%6
		for set != 0 {
			square := bits.LeadingZeros64(set)
			set &^= uint64(1) << uint(63-square)
			var destinations uint64
			switch kind {
			case 0:
				destinations = exchangeGeometry.kings[square]
			case 1, 2, 3:
				destinations = evaluationSliders(square, occupied, kind != 2, kind != 3)
			case 4:
				destinations = exchangeGeometry.knights[square]
			case 5:
				destinations = evaluationPawnAttacks(uint64(1)<<uint(63-square), side)
			}
			attacks[side][kind] |= destinations
			twice[side] |= all[side] & destinations
			all[side] |= destinations
			if kind >= 1 && kind <= 4 {
				count := bits.OnesCount64(destinations & rings[1-side])
				if count != 0 {
					attackers[side]++
					weight := 2
					if kind == 1 {
						weight = 4
					}
					if kind == 2 {
						weight = 3
					}
					ringUnits[side] += count * weight
				}
			}
		}
	}
	var danger [2]int
	for side := 0; side < 2; side++ {
		enemy := 1 - side
		king := kings[enemy]
		safe := (^all[enemy] | (twice[side] &^ attacks[enemy][5] &^ twice[enemy])) &^ occupancy[side]
		orthogonal := evaluationSliders(king, occupied, false, true)
		diagonal := evaluationSliders(king, occupied, true, false)
		checks := 24 * min(4, bits.OnesCount64(exchangeGeometry.knights[king]&attacks[side][4]&safe))
		checks += 16 * min(4, bits.OnesCount64(diagonal&attacks[side][3]&safe))
		checks += 24 * min(4, bits.OnesCount64(orthogonal&attacks[side][2]&safe))
		checks += 12 * min(4, bits.OnesCount64((orthogonal|diagonal)&attacks[side][1]&safe))
		pressure := (ringUnits[side] + checks) * min(4, max(1, attackers[side])) / 2
		if pieces[side*6+1] == 0 {
			pressure /= 4
		}
		resource := 4*bits.OnesCount64(pieces[side*6+1]) + 2*bits.OnesCount64(pieces[side*6+2]) + bits.OnesCount64(pieces[side*6+3]|pieces[side*6+4])
		shelter := 0
		for file := max(0, king%8-1); file <= min(7, king%8+1); file++ {
			cost := 20
			step := 8
			if enemy == 1 {
				step = -8
			}
			for advance := 1; advance <= 3; advance++ {
				sq := king/8*8 + file + advance*step
				if sq < 0 || sq >= 64 {
					break
				}
				if pieces[enemy*6+5]&(uint64(1)<<uint(63-sq)) != 0 {
					cost = (advance - 1) * 6
					break
				}
			}
			shelter += cost
		}
		danger[enemy] = min(600, pressure+shelter*min(12, resource)/12) * phase / 24
	}
	return danger[1] - danger[0]
}

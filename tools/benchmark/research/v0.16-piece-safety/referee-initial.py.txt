"""Offline, finite-node assessment of every tied best fixed-depth move and timed choice.
Run with lichess-bot/venv/bin/python. Does not open a network connection.
"""
import hashlib,json,sys,time
from pathlib import Path
import chess,chess.engine
root=Path(__file__).resolve().parent
inputs=json.loads((root/'probe-inputs.json').read_text())['cases']
byid={(c['game'],c['ply']):c for c in inputs}
paths=sys.argv[1:] or ['baseline-probes.json','candidate-probes.json']
queries={}
for name in paths:
 for r in json.loads((root/name).read_text()):
  if r['Complete'] and r['Legal']:
   for m in r['Moves']:queries[(r['Game'],r['Ply'],m)]=None
out=[]
engine=chess.engine.SimpleEngine.popen_uci('/opt/homebrew/bin/stockfish',timeout=30)
engine.configure({'Threads':1,'Hash':64,'Skill Level':20,'UCI_LimitStrength':False})
metadata={'method':'Every tied fixed-depth best and every timed selection from named files, unrestricted finite-node comparator plus each legal choice with equal 100000-node budget and fresh hash. Full history. Not ground truth, not tuning. Mate kept separate. No unique-reference-move requirement.','inputs':paths,'nodes':100000,'id':engine.id,'stockfish_sha256':hashlib.sha256(Path('/opt/homebrew/bin/stockfish').read_bytes()).hexdigest(),'results':out}
try:
 for game,ply in sorted(set((g,p) for g,p,_ in queries)):
  c=byid[game,ply];cmd=c['position'].split();i=cmd.index('moves') if 'moves' in cmd else len(cmd)
  board=chess.Board(' '.join(cmd[2:i])) if cmd[1]=='fen' else chess.Board()
  for m in cmd[i+1:]:board.push_uci(m)
  assert board.fen(en_passant='fen').split()[:4]==c['fen'].split()[:4],(game,board.fen(),c['fen'])
  for move in [None]+sorted(m for g,p,m in queries if (g,p)==(game,ply)):
   engine.protocol.send_line('ucinewgame');engine.configure({'Clear Hash':None})
   info=engine.analyse(board,chess.engine.Limit(nodes=100000),root_moves=[chess.Move.from_uci(move)] if move else None,game=object())
   score=info['score'].pov(board.turn)
   out.append({'game':game,'ply':ply,'move':move,'score':{'mate':score.mate()} if score.is_mate() else {'cp':score.score()},'depth':info.get('depth'),'nodes':info.get('nodes'),'pv':[m.uci() for m in info.get('pv',[])]})
  (root/'referee-results.json').write_text(json.dumps(metadata,indent=2)+'\n')
finally:engine.quit()
print(f'Saved {len(out)} referee queries, including all legal alternatives selected by either engine.')

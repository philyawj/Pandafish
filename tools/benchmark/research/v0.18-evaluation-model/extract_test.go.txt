package main
import("encoding/json";"os";"testing";"github.com/corentings/chess/v2")
func TestExportEvaluationFeatures(t *testing.T){
 data,err:=os.ReadFile(os.Getenv("MODEL_INPUT"));if err!=nil{t.Fatal(err)}
 var input struct{Cases []struct{ID,FEN string}};if err=json.Unmarshal(data,&input);err!=nil{t.Fatal(err)}
 type row struct{ID,FEN string;Base,Quiet int;Features chess.EvalFeatures}
 out:=[]row{}
 for _,c:=range input.Cases{p:=evalFEN(t,c.FEN);q:=quiescence(p,quiescenceDepth,-searchInfinity,searchInfinity);if p.Turn()==chess.Black{q=-q};out=append(out,row{c.ID,c.FEN,baseStaticScore(p,chess.White),q,p.Board().EvaluationFeatures()})}
 encoded,_:=json.MarshalIndent(out,"","  ");if err=os.WriteFile(os.Getenv("MODEL_OUTPUT"),append(encoded,'\n'),0644);err!=nil{t.Fatal(err)}
}

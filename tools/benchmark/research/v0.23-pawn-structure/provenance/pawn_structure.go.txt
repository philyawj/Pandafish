package main

import "math/bits"

const pawnFileA uint64 = 0x8080808080808080
const pawnFileH uint64 = 0x0101010101010101

// Pawn relations are positional hints, not proofs of mobility or promotion.
// Masks use the library's A1-high-bit convention. Only the immediate square
// ahead discounts a blocked passer; pins and king blockade are not evaluated.
func pawnStructure(pieces [12]uint64, phase int) int {
	pawns := [2]uint64{pieces[5], pieces[11]}
	occupied := uint64(0)
	for _, set := range pieces {
		occupied |= set
	}
	var mg, eg [2]int
	for side, own := range pawns {
		enemy := pawns[1-side]
		files := own
		files |= files << 8
		files |= files << 16
		files |= files << 32
		doubled := bits.OnesCount64(own) - bits.OnesCount64(files&0xff00000000000000)
		files |= files >> 8
		files |= files >> 16
		files |= files >> 32
		neighbors := ((files &^ pawnFileA) << 1) | ((files &^ pawnFileH) >> 1)
		isolated := bits.OnesCount64(own &^ neighbors)
		mg[side] -= 12*doubled + 10*isolated
		eg[side] -= 20*doubled + 16*isolated
		front := enemy | ((enemy &^ pawnFileA) << 1) | ((enemy &^ pawnFileH) >> 1)
		blocked := occupied << 8
		if side == 0 {
			front <<= 8
			front |= front << 8
			front |= front << 16
			front |= front << 32
		} else {
			front >>= 8
			front |= front >> 8
			front |= front >> 16
			front |= front >> 32
			blocked = occupied >> 8
		}
		passers := own &^ front
		middle := [8]int{0, 0, 5, 12, 25, 45, 75, 0}
		ending := [8]int{0, 0, 10, 25, 50, 90, 150, 0}
		for passers != 0 {
			square := bits.LeadingZeros64(passers)
			bit := uint64(1) << uint(63-square)
			passers &^= bit
			rank := square / 8
			if side == 1 {
				rank = 7 - rank
			}
			m, e := middle[rank], ending[rank]
			if bit&blocked != 0 {
				m /= 2
				e /= 2
			}
			mg[side] += m
			eg[side] += e
		}
	}
	return ((mg[0]-mg[1])*phase + (eg[0]-eg[1])*(24-phase)) / 24
}

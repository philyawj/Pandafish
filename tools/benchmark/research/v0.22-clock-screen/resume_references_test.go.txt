package main

import (
	"encoding/json"
	"os"
	"path/filepath"
	"testing"
)

// Reporting continuation only: probeReferences labels saved native choices and
// never launches a baseline/candidate engine or rewrites their observations.
func TestResumeReferencesWithoutNative(t *testing.T) {
	out := os.Getenv("RESUME_OUT")
	if out == "" {
		t.Skip("explicit interrupted run required")
	}
	teacher, cache := os.Getenv("RESUME_TEACHER"), os.Getenv("RESUME_CACHE")
	var method struct {
		IDs   map[string]string `json:"identities_sha256"`
		MS    int               `json:"milliseconds"`
		Nodes int               `json:"reference_nodes"`
	}
	if e := load(filepath.Join(out, "method.json"), &method); e != nil {
		t.Fatal(e)
	}
	h, e := probeDigest(teacher)
	if e != nil || h != method.IDs["reference"] || method.Nodes != 100000 || method.MS != 100 {
		t.Fatal("resume identity", e)
	}
	var native []probeNative
	if e = load(filepath.Join(out, "native.json"), &native); e != nil {
		t.Fatal(e)
	}
	if len(native) != 84 {
		t.Fatal("native cases", len(native))
	}
	trials := 0
	for _, r := range native {
		if r.Error != "" || len(r.Trials) != 6 {
			t.Fatal("native error/shape", r.Case.ID, r.Error)
		}
		for _, v := range r.Trials {
			if v.Query.Error != "" {
				t.Fatal("saved query error", r.Case.ID, v.Query.Error)
			}
			trials++
		}
	}
	before := map[string]string{}
	for _, name := range []string{"native.json", "native.jsonl", "method.json"} {
		h, e := probeDigest(filepath.Join(out, name))
		if e != nil {
			t.Fatal(e)
		}
		before[name] = h
	}
	err := probeReferences(teacher, out, method.MS, native, cache, method.IDs["reference"], false)
	after := map[string]string{}
	for name, want := range before {
		h, e := probeDigest(filepath.Join(out, name))
		if e != nil || h != want {
			t.Fatal("native observation changed", name, e)
		}
		after[name] = h
	}
	data, e := json.MarshalIndent(map[string]any{"reference_error": func() string {
		if err != nil {
			return err.Error()
		}
		return ""
	}(), "native_cases": len(native), "native_queries": trials, "before_sha256": before, "after_sha256": after, "native_unchanged": true, "cache_directory": cache}, "", "  ")
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile(filepath.Join(out, "reference-repair.json"), append(data, '\n'), 0644); e != nil {
		t.Fatal(e)
	}
	if err != nil {
		t.Fatal(err)
	}
}

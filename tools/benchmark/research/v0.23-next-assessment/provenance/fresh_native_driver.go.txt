package main
import("encoding/json";"fmt";"os";"os/exec";"path/filepath";"time")
func main(){if len(os.Args)!=3{panic("evidence directory and candidate binary required")};dir,candidate:=os.Args[1],os.Args[2];checks:=[]map[string]any{}
 for _,s:=range []struct{name string;ms,trials int}{{"fresh-100",100,3},{"fresh-1000",1000,1}} {args:=[]string{"choices","--baseline","tools/benchmark/bin/pandafish-v0.22","--candidate",candidate,"--teacher","/opt/homebrew/bin/stockfish","--input",filepath.Join(dir,"fresh-generation/inputs.json"),"--out",filepath.Join(dir,s.name),"--milliseconds",fmt.Sprint(s.ms),"--trials",fmt.Sprint(s.trials),"--workers","3","--by-turn"};if s.ms==1000{args=append(args,"--cache",filepath.Join(dir,"fresh-100"))};cmd:=exec.Command("tools/benchmark/bin/research",args...);log,e:=os.OpenFile(filepath.Join(dir,s.name+"-run.txt"),os.O_CREATE|os.O_EXCL|os.O_WRONLY,0644);if e!=nil{panic(e)};cmd.Stdout=log;cmd.Stderr=log;start:=time.Now();err:=cmd.Run();log.Close();q:=struct{Passed bool `json:"passed"`}{ };b,e:=os.ReadFile(filepath.Join(dir,s.name,"quality.json"));if e==nil{e=json.Unmarshal(b,&q)};status:=0;if err!=nil||e!=nil{status=1};checks=append(checks,map[string]any{"suite":s.name,"command":cmd.Args,"exit":status,"quality_passed":q.Passed,"seconds":time.Since(start).Seconds()});b,e=json.MarshalIndent(checks,"","  ");if e!=nil{panic(e)};if e=os.WriteFile(filepath.Join(dir,"fresh-native-checks.json"),append(b,'\n'),0644);e!=nil{panic(e)};fmt.Println(s.name,status,q.Passed);if status!=0||!q.Passed{os.Exit(1)}}
}

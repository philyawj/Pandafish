// Remove only unchanged base-tag metadata that the shared replay format rejects.
package main
import("crypto/sha256";"encoding/json";"fmt";"os";"os/exec";"path/filepath";"strings")
func main(){for _,dir:=range os.Args[1:]{path:=filepath.Join(dir,"candidate-source.json");b,e:=os.ReadFile(path);if e!=nil{panic(e)};original:=sha256.Sum256(b);var x struct{Base string `json:"base_tag"`;Files map[string]string `json:"files"`};if e=json.Unmarshal(b,&x);e!=nil{panic(e)};omitted:=map[string]string{};for p,content:=range x.Files {if strings.HasSuffix(p,".go")||strings.HasSuffix(p,".json")||p=="go.mod"||p=="go.sum"{continue};base,e:=exec.Command("git","show",x.Base+":"+p).Output();if e!=nil{panic(e)};if string(base)!=content{panic("cannot omit changed file: "+p)};h:=sha256.Sum256(base);omitted[p]=fmt.Sprintf("%x",h);delete(x.Files,p)};b,e=json.MarshalIndent(x,"","  ");if e!=nil{panic(e)};b=append(b,'\n');if e=os.WriteFile(filepath.Join(dir,"candidate-replay-source.json"),b,0644);e!=nil{panic(e)};updated:=sha256.Sum256(b);report,e:=json.MarshalIndent(map[string]any{"original_sha256":fmt.Sprintf("%x",original),"replay_sha256":fmt.Sprintf("%x",updated),"retained_files":len(x.Files),"omitted_unchanged_base_files":omitted,"runtime_source_changed":false},"","  ");if e!=nil{panic(e)};if e=os.WriteFile(filepath.Join(dir,"source-format-repair.json"),append(report,'\n'),0644);e!=nil{panic(e)};fmt.Println(dir,"retained",len(x.Files),"omitted unchanged",len(omitted))}}

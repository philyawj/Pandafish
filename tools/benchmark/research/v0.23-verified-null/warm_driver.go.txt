// Alternate isolated fixed-depth runs. Selective scores may differ; all work counts.
package main

import("encoding/json";"fmt";"os";"os/exec";"path/filepath";"sort")
type row struct{Game,Ply,Depth,Score int;Best,FEN string;Nodes uint64;ParseMS,SearchMS float64}
func main(){repo,e:=os.Getwd();if e!=nil{panic(e)};dir:=filepath.Join(repo,"tools/benchmark/research/v0.23-verified-null");input:=filepath.Join(repo,"tools/benchmark/research/v0.21-rank-order/screen/v0.21-rank-order-dev-vs-v0.20-and-stockfish-3-and-sacrifice-v0.2.0-b528f0ba");checks:=[]map[string]any{}
 put:=func(name string,v any){b,e:=json.MarshalIndent(v,"","  ");if e!=nil{panic(e)};if e=os.WriteFile(filepath.Join(dir,name),append(b,'\n'),0644);e!=nil{panic(e)}}
 ratios,nratios:=[]float64{},[]float64{};issues:=[]string{}
 for pair:=1;pair<=3;pair++{order:=[]string{"baseline","candidate"};if pair%2==0{order[0],order[1]=order[1],order[0]};data:=map[string][]row{}
  for _,actor:=range order {name:=fmt.Sprintf("warm-%s-pair-%d",actor,pair);path:=filepath.Join(dir,name+".json");if _,e:=os.Stat(path);!os.IsNotExist(e){panic("refuse overwrite")};binary:="/tmp/pandafish-v023-baseline.test";if actor=="candidate"{binary="/tmp/pandafish-v023-null.test"};cmd:=exec.Command(binary,"-test.run=^TestWarmPipelineProfile$","-test.count=1","-test.v");cmd.Dir=repo;cmd.Env=append(os.Environ(),"WARM_INPUT="+input,"WARM_OUTPUT="+path,"WARM_MODE=search","WARM_DEPTH=5");b,err:=cmd.CombinedOutput();if e=os.WriteFile(filepath.Join(dir,name+".txt"),b,0644);e!=nil{panic(e)};status:=0;if err!=nil{status=1};checks=append(checks,map[string]any{"actor":actor,"pair":pair,"command":cmd.Args,"exit":status});put("warm-checks.json",checks);if err!=nil{panic(err)};b,e=os.ReadFile(path);if e!=nil{panic(e)};var x struct{Rows []row};if e=json.Unmarshal(b,&x);e!=nil{panic(e)};data[actor]=x.Rows;fmt.Println(name,len(x.Rows))}
  a,b:=data["baseline"],data["candidate"];if len(a)!=1145||len(a)!=len(b){panic("warm coverage")};am,bm:=0.,0.;var an,bn uint64;changed:=0;for i,x:=range a{y:=b[i];am+=x.SearchMS;bm+=y.SearchMS;an+=x.Nodes;bn+=y.Nodes;if x.Game!=y.Game||x.Ply!=y.Ply||x.FEN!=y.FEN||x.Depth!=y.Depth||y.Best=="0000"{issues=append(issues,fmt.Sprintf("pair%d/%d/%d",pair,x.Game,x.Ply))};if x.Score!=y.Score||x.Best!=y.Best{changed++}};ratio:=bm/am;nr:=float64(bn)/float64(an);ratios=append(ratios,ratio);nratios=append(nratios,nr);put(fmt.Sprintf("warm-pair-%d-summary.json",pair),map[string]any{"baseline_ms":am,"candidate_ms":bm,"ratio":ratio,"baseline_nodes":an,"candidate_nodes":bn,"node_ratio":nr,"roots":len(a),"changed_roots":changed,"issues":issues});fmt.Printf("pair%d baseline%.1f candidate%.1f ratio%.5f nodes%.5f\n",pair,am,bm,ratio,nr)
 }
 sort.Float64s(ratios);sort.Float64s(nratios);passed:=len(issues)==0&&ratios[1]<=.85&&nratios[2]<=.8;put("warm-analysis.json",map[string]any{"ratios":ratios,"node_ratios":nratios,"median_ratio":ratios[1],"saving_percent":(1-ratios[1])*100,"coverage_issues":issues,"opportunity_gate_passed":passed});if !passed{os.Exit(1)}
}

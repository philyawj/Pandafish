func TestVerifiedNullControls(t *testing.T) {
	for _, c := range []struct {
		name, fen     string
		depth, beta   int
		checked, want bool
	}{
		{"material advantage", "rnb1kbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1", 4, 500, false, true},
		{"white quiet mate threat", "8/8/8/8/8/5kq1/8/Q6K w - - 0 1", 4, -100, false, false},
		{"black quiet mate threat", "q6k/8/5KQ1/8/8/8/8/8 b - - 0 1", 4, -100, false, false},
		{"shallow", "rnb1kbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1", 3, 500, false, false},
		{"checked", "rnb1kbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1", 4, 500, true, false},
		{"mate window", "rnb1kbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1", 4, -mateBound, false, false},
		{"pawn ending", "7k/7p/8/8/8/8/P7/K7 w - - 0 1", 4, -500, false, false},
		{"low phase", "7k/8/8/8/8/8/8/KR6 w - - 0 1", 4, -500, false, false},
		{"near promotion", "rnb1kbnr/Pppppppp/8/8/8/8/1PPPPPPP/RNBQKBNR w KQkq - 0 1", 4, -500, false, false},
		{"draw clock", "rnb1kbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 80 1", 4, 500, false, false},
		{"static below beta", "rnb1kbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1", 4, 2000, false, false},
	} {
		t.Run(c.name, func(t *testing.T) {
			p := evalFEN(t, c.fen)
			fen, hash := p.String(), p.ZobristHash()
			w := searcher{table: newTable()}
			w.pushKey(123)
			w.pushPosition(p)
			counts, key := maps.Clone(w.repetitions), w.historyKey
			cut, done := w.verifiedNullCut(p, c.depth, c.beta, c.checked, 0)
			if !done || cut != c.want {
				t.Fatalf("cut=%v complete=%v nodes=%d", cut, done, w.nodes)
			}
			if !maps.Equal(counts, w.repetitions) || key != w.historyKey || p.String() != fen || p.ZobristHash() != hash {
				t.Fatal("real state changed")
			}
			if _, hit := w.table.probe(p, key); hit || w.table.boardHint(p) != 0 {
				t.Fatal("probe polluted real table")
			}
			if c.want && w.nodes == 0 {
				t.Fatal("probe work omitted")
			}
		})
	}
}

func TestVerifiedNullCancellationAndDrawPriority(t *testing.T) {
	p := chess.StartingPosition()
	ctx, cancel := context.WithCancel(context.Background())
	cancel()
	w := searcher{ctx: ctx}
	w.pushPosition(p)
	if _, done := w.verifiedNullCut(p, 4, -500, false, 0); done {
		t.Fatal("canceled result completed")
	}
	w = searcher{deadline: time.Unix(10, 0), now: func() time.Time { return time.Unix(11, 0) }}
	w.pushPosition(p)
	if _, done := w.verifiedNullCut(p, 4, -500, false, 0); done {
		t.Fatal("deadline ignored")
	}
	w = searcher{}
	w.pushPosition(p)
	w.pushPosition(p)
	if score, done := w.search(p, 4, -501, -500, false); !done || score != 0 || w.nodes != 1 {
		t.Fatal("repetition did not precede selective pruning", score, done, w.nodes)
	}
}

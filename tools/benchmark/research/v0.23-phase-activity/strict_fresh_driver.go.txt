package main
import("encoding/json";"fmt";"os";"os/exec";"path/filepath";"time")
func main(){dir:="tools/benchmark/research/v0.23-phase-activity";candidate:="tools/benchmark/bin/pandafish-v023-phase-activity";checks:=[]map[string]any{}
 for _,s:=range []struct{name string;ms,trials int}{{"fresh-100",100,3},{"fresh-1000",1000,1}} {args:=[]string{"choices","--baseline","tools/benchmark/bin/pandafish-v0.22","--candidate",candidate,"--teacher","/opt/homebrew/bin/stockfish","--input",filepath.Join(dir,"fresh-generation/inputs.json"),"--out",filepath.Join(dir,s.name),"--milliseconds",fmt.Sprint(s.ms),"--trials",fmt.Sprint(s.trials),"--workers","3","--by-turn"};if s.ms==1000{args=append(args,"--cache",filepath.Join(dir,"fresh-100"))};cmd:=exec.Command("tools/benchmark/bin/research",args...);log,e:=os.OpenFile(filepath.Join(dir,s.name+"-run.txt"),os.O_CREATE|os.O_EXCL|os.O_WRONLY,0644);if e!=nil{panic(e)};cmd.Stdout=log;cmd.Stderr=log;start:=time.Now();err:=cmd.Run();log.Close();q:=struct{Passed bool `json:"passed"`;Mean map[string]float64 `json:"mean_loss_cp"`;Severe map[string]int `json:"severe_over_300"`}{ };b,e:=os.ReadFile(filepath.Join(dir,s.name,"quality.json"));if e==nil{e=json.Unmarshal(b,&q)};status:=0;if err!=nil||e!=nil{status=1};strict:=status==0&&q.Passed&&q.Mean["baseline"]>0&&q.Mean["candidate"]<=.95*q.Mean["baseline"]&&q.Severe["candidate"]<=q.Severe["baseline"];checks=append(checks,map[string]any{"suite":s.name,"command":cmd.Args,"exit":status,"original_quality_passed":q.Passed,"stricter_independent_gate_passed":strict,"mean_loss_cp":q.Mean,"severe_over_300":q.Severe,"seconds":time.Since(start).Seconds()});b,e=json.MarshalIndent(checks,"","  ");if e!=nil{panic(e)};if e=os.WriteFile(filepath.Join(dir,"strict-fresh-checks.json"),append(b,'\n'),0644);e!=nil{panic(e)};fmt.Println(s.name,status,"original",q.Passed,"strict",strict);if !strict{os.Exit(1)}}
}

package main
import("context";"runtime";"strconv";"encoding/json";"os";"strings";"testing";"time";"github.com/corentings/chess/v2")
type architectureCase struct{ID,Group,Position string;NextTwo []string `json:"next_two"`}
type architectureResult struct{ID,Group,Mode string;Depth,Score int;Best []string;Nodes uint64;MS float64;Complete,Legal bool}
func architectureInputs(t testing.TB)[]architectureCase{t.Helper();b,e:=os.ReadFile(os.Getenv("ARCH_INPUT"));if e!=nil{t.Fatal(e)};var d struct{Cases []architectureCase};if e=json.Unmarshal(b,&d);e!=nil{t.Fatal(e)};return d.Cases}
func architectureGame(t testing.TB,c architectureCase)*chess.Game{t.Helper();g,e:=parsePosition(strings.Fields(strings.TrimPrefix(c.Position,"position ")));if e!=nil{t.Fatal(e)};return g}
func architectureWorker(g *chess.Game,tt *transpositionTable)*searcher{w:=&searcher{table:tt,killers:new(killerTable),deadline:time.Now().Add(10*time.Second)};for _,k:=range repetitionHistory(g){w.pushKey(k)};return w}
func architectureFixed(t testing.TB,c architectureCase,g *chess.Game,w *searcher,depth int,mode string)architectureResult{t.Helper();w.nodes=0;w.deadline=time.Now().Add(10*time.Second);start:=time.Now();best,score,ok:=w.root(g.Position(),rootMoves(g.Position(),nil),depth);legal:=len(best)>0;for _,s:=range best{found:=false;for _,m:=range g.Position().ValidMoves(){if m.String()==s{found=true;break}};legal=legal&&found};return architectureResult{c.ID,c.Group,mode,depth,score,best,w.nodes,float64(time.Since(start))/float64(time.Millisecond),ok,legal}}
func TestArchitectureProbe(t *testing.T){if os.Getenv("ARCH_OUTPUT")==""{t.Skip("diagnostic requires explicit output path")};
 cases:=architectureInputs(t);var out []architectureResult;save:=func(){b,_:=json.MarshalIndent(out,"","  ");if e:=os.WriteFile(os.Getenv("ARCH_OUTPUT"),append(b,'\n'),0644);e!=nil{t.Fatal(e)}}
 for _,c:=range cases{g:=architectureGame(t,c);switch os.Getenv("ARCH_MODE"){
 case "fixed":depths:=[]int{3,4};if ds:=os.Getenv("ARCH_DEPTHS");ds!=""{depths=nil;for _,x:=range strings.Split(ds,","){n,e:=strconv.Atoi(x);if e!=nil{t.Fatal(e)};depths=append(depths,n)}};for _,d:=range depths{out=append(out,architectureFixed(t,c,g,architectureWorker(g,newTable()),d,"cold"))}
 case "timed":for trial:=0;trial<3;trial++{limits:=goLimits{moveTime:100*time.Millisecond,hasTime:true};budget,_:=limits.budget(g.Position().Turn());start:=time.Now();result:=iterativeSearchWithHistory(context.Background(),g.Position(),limits,budget,start,newTable(),repetitionHistory(g));legal:=false;for _,m:=range g.Position().ValidMoves(){if m.String()==result.move{legal=true}};out=append(out,architectureResult{c.ID,c.Group,"100ms",result.depth,result.score,[]string{result.move},result.metrics.Nodes,float64(time.Since(start))/float64(time.Millisecond),result.depth>0,legal})}
 case "warm":if len(c.NextTwo)!=2{continue};tt:=newTable();w:=architectureWorker(g,tt);out=append(out,architectureFixed(t,c,g,w,4,"cold-parent"));out=append(out,architectureFixed(t,c,g,architectureWorker(g,tt),4,"repeat-parent"));s:=c.Position;if !strings.Contains(s," moves "){s+=" moves"};next:=c;next.Position=s+" "+strings.Join(c.NextTwo," ");ng:=architectureGame(t,next);out=append(out,architectureFixed(t,c,ng,architectureWorker(ng,tt),4,"warm-next"));out=append(out,architectureFixed(t,c,ng,architectureWorker(ng,newTable()),4,"cold-next"))
 };save()}
}

func BenchmarkArchitectureProfile(b *testing.B){if os.Getenv("ARCH_NO_MEM_PROFILE")=="1"{runtime.MemProfileRate=0};cases:=architectureInputs(b)[:24];games:=make([]*chess.Game,len(cases));for i,c:=range cases{games[i]=architectureGame(b,c)};tt:=newTable();b.ReportAllocs();b.ResetTimer();for i:=0;i<b.N;i++{tt.clear();g:=games[i%len(games)];limits:=goLimits{moveTime:100*time.Millisecond,hasTime:true};budget,_:=limits.budget(g.Position().Turn());ctx,cancel:=context.WithCancel(context.Background());iterativeSearchWithHistory(ctx,g.Position(),limits,budget,time.Now(),tt,repetitionHistory(g));cancel()}}

"""Offline scheduling diagnostic. Does not execute workers or make network calls."""
import argparse
import ast
import __future__
import hashlib
import json
import logging
from pathlib import Path
import queue
import requests
import sys
from unittest.mock import patch

parser = argparse.ArgumentParser()
parser.add_argument('--client', required=True)
parser.add_argument('--upstream-source', required=True)
parser.add_argument('--output', required=True)
args = parser.parse_args()
sys.path.insert(0, args.client)
from lib import lichess_bot as bot


class Pool:
    def __init__(self):
        self.jobs = []

    def apply_async(self, function, kwds, error_callback):
        self.jobs.append(dict(kwds))  # Snapshot only. No worker execution.


def namespace(path):
    source = Path(path).read_text()
    tree = ast.parse(source)
    names = {'start_game_thread', 'watch_control_stream'}
    selected = ast.Module(body=[n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name in names], type_ignores=[])
    assert len(selected.body) == len(names)
    ns = dict(vars(bot))
    exec(compile(selected, str(path), 'exec', flags=__future__.annotations.compiler_flag), ns)
    return ns, hashlib.sha256(source.encode()).hexdigest()


def starts(ns, reserved):
    pool = Pool()
    active = {'samegame': 'opponent'} if reserved else {}
    running = set()
    kwargs = dict(control_queue=queue.Queue(), pgn_queue=queue.Queue(), li=None)
    start = ns['start_game_thread']
    extra = [running] if 'games_in_progress' in start.__code__.co_varnames[:start.__code__.co_argcount] else []
    start(active, 'samegame', 'opponent', kwargs, pool, *extra)
    first = len(pool.jobs)
    start(active, 'samegame', 'opponent', kwargs, pool, *extra)
    result = dict(pre_reserved=reserved, jobs_after_first=first, jobs_after_duplicate=len(pool.jobs), logical_active_ids=len(active))
    # Models the existing local_game_done pop, not a real worker completion.
    active.pop('samegame', None)
    result['modeled_jobs_remaining_after_one_completion'] = max(0, len(pool.jobs)-1)
    result['modeled_logical_active_ids_after_one_completion'] = len(active)
    return result


def reconnect(ns):
    class Response:
        def __init__(self, number):
            self.number = number

        def __enter__(self):
            return self

        def __exit__(self, *ignored):
            return False

        def iter_lines(self):
            yield json.dumps({'type': 'gameStart', 'game': {'id': 'samegame'}}).encode()
            if self.number == 1:
                raise requests.exceptions.ChunkedEncodingError('simulated premature end')
            bot.stop.terminated = True

    class Client:
        calls = 0

        def get_event_stream(self):
            self.calls += 1
            return Response(self.calls)

    output = queue.Queue()
    client = Client()
    bot.stop.terminated = False
    with patch.object(bot.time, 'sleep'), patch.object(bot.logger, 'warning') as warning:
        ns['watch_control_stream'](output, client)
    events = list(output.queue)
    return dict(stream_connections=client.calls, game_start_events=sum(e['type']=='gameStart' for e in events), reconnect_warnings=warning.call_count, note='Synthetic server deliberately replays the start. This does not establish what the real SSE server emitted.')


logging.disable(logging.CRITICAL)
result = {'scope': 'Actual selected function bodies, mocked pool and event stream; no workers, engines, online client, or API calls.', 'clients': {}}
with patch.object(requests.sessions.Session, 'request', side_effect=AssertionError('Network forbidden')):
    for label, path in [('installed', Path(args.client)/'lib/lichess_bot.py'), ('upstream', Path(args.upstream_source))]:
        ns, digest = namespace(path)
        cases = [starts(ns, False), starts(ns, True)]
        expected = 2 if label == 'installed' else 1
        assert all(c['jobs_after_first']==1 and c['jobs_after_duplicate']==expected for c in cases)
        stream = reconnect(ns)
        assert stream['game_start_events']==2 and stream['reconnect_warnings']==1
        result['clients'][label] = dict(source_sha256=digest, cases=cases, reconnect=stream)
Path(args.output).write_text(json.dumps(result, indent=2)+'\n')
print('Verified: installed queues duplicate workers; current upstream suppresses the duplicate and starts reserved games normally. No client changes or network calls.')

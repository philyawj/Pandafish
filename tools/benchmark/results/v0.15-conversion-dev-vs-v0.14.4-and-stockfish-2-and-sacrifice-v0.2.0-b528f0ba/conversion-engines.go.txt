package main

import (
	"bufio"
	"context"
	"crypto/sha256"
	"encoding/hex"
	"fmt"
	"github.com/corentings/chess/v2"
	"os"
	"os/exec"
	"strings"
	"time"
)

type position struct {
	Game      string `json:"game"`
	Ply       int    `json:"ply"`
	Split     string `json:"split"`
	FEN       string `json:"fen"`
	Position  string `json:"position"`
	Reference string `json:"reference"`
	Played    string `json:"played"`
}
type probe struct {
	Request        string   `json:"request"`
	Engine         string   `json:"engine"`
	BinarySHA      string   `json:"binary_sha256"`
	Move           string   `json:"move"`
	Legal          bool     `json:"legal"`
	ExactReference bool     `json:"exact_reference"`
	Output         []string `json:"uci_output"`
	Error          string   `json:"error,omitempty"`
}
type audited struct {
	position
	Probes []probe `json:"probes"`
}

func hash(path string) string {
	data, err := os.ReadFile(path)
	if err != nil {
		panic(err)
	}
	h := sha256.Sum256(data)
	return hex.EncodeToString(h[:])
}
func search(path string, p position, request string) probe {
	ctx, cancel := context.WithTimeout(context.Background(), 20*time.Second)
	defer cancel()
	cmd := exec.CommandContext(ctx, path)
	in, err := cmd.StdinPipe()
	if err != nil {
		panic(err)
	}
	out, err := cmd.StdoutPipe()
	if err != nil {
		panic(err)
	}
	if err = cmd.Start(); err != nil {
		panic(err)
	}
	defer func() { in.Close(); cmd.Process.Kill(); cmd.Wait() }()
	scanner := bufio.NewScanner(out)
	wait := func(marker string) []string {
		var lines []string
		for scanner.Scan() {
			s := scanner.Text()
			lines = append(lines, s)
			if strings.HasPrefix(s, marker) {
				return lines
			}
		}
		panic("missing UCI response: " + marker)
	}
	fmt.Fprintln(in, "uci")
	handshake := wait("uciok")
	r := probe{BinarySHA: hash(path), Request: request}
	for _, line := range handshake {
		if strings.HasPrefix(line, "id name ") {
			r.Engine = strings.TrimPrefix(line, "id name ")
		}
	}
	fmt.Fprintln(in, "ucinewgame\nisready")
	wait("readyok")
	fmt.Fprintln(in, p.Position)
	fmt.Fprintln(in, request)
	r.Output = wait("bestmove ")
	r.Move = strings.Fields(r.Output[len(r.Output)-1])[1]
	fen, err := chess.FEN(p.FEN)
	if err != nil {
		panic(err)
	}
	g := chess.NewGame(fen)
	for _, m := range g.Position().ValidMoves() {
		if m.String() == r.Move {
			r.Legal = true
		}
	}
	if !r.Legal {
		panic("illegal audit move: " + r.Move)
	}
	r.ExactReference = r.Move == p.Reference
	fmt.Fprintln(in, "quit")
	return r
}

package main
import("bytes";"crypto/sha256";"encoding/json";"fmt";"os";"path/filepath")

 result:=map[string]any{"passed":true,"candidate_source_sha256":fmt.Sprintf("%x",sha256.Sum256(b)),"core_files":len(checked),"only_version_substitutions":[]string{"main.go","main_test.go"},"current_version":"v1.0","source_sha256":checked};b,e=json.MarshalIndent(result,"","  ");must(e);must(os.WriteFile(os.Args[1],append(b,'\n'),0644));fmt.Printf("Version-only source boundary passes on %d files\n",len(checked))}
